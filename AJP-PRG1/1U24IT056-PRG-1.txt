class Point {
    private final double x;
    private final double y;

    public Point(double x, double y) {
        this.x = x;
        this.y = y;
    }

    public double getX() {
        return x;
    }

    public double getY() {
        return y;
    }

    public double distanceTo(Point p) {
        double dx = x - p.getX();
        double dy = y - p.getY();
        return Math.sqrt(dx * dx + dy * dy);
    }
}

class Quadrilateral {
    protected final Point point1;
    protected final Point point2;
    protected final Point point3;
    protected final Point point4;

    public Quadrilateral(Point p1, Point p2, Point p3, Point p4) {
        this.point1 = p1;
        this.point2 = p2;
        this.point3 = p3;
        this.point4 = p4;
    }
}

class Trapezoid extends Quadrilateral {

    public Trapezoid(Point p1, Point p2, Point p3, Point p4) {
        super(p1, p2, p3, p4);
    }

    public double calculateArea() {
        double base1 = point1.distanceTo(point2);
        double base2 = point3.distanceTo(point4);

        if (base1 == 0) {
            throw new IllegalArgumentException("Base cannot be zero");
        }

        // Perpendicular height from point3 to the line p1-p2
        double dx = point2.getX() - point1.getX();
        double dy = point2.getY() - point1.getY();

        double height = Math.abs(
            dx * (point3.getY() - point1.getY())
            - dy * (point3.getX() - point1.getX())
        ) / base1;

        return (base1 + base2) * height / 2.0;
    }
}

class Rectangle extends Quadrilateral {

    public Rectangle(Point p1, Point p2, Point p3, Point p4) {
        super(p1, p2, p3, p4);
    }

    public double calculateArea() {
        double length = point1.distanceTo(point2);
        double width = point2.distanceTo(point3);

        return length * width;
    }
}

class Square extends Rectangle {

    public Square(Point p1, double sideLength) {
        super(
            p1,
            new Point(p1.getX() + sideLength, p1.getY()),
            new Point(p1.getX() + sideLength,
                      p1.getY() + sideLength),
            new Point(p1.getX(), p1.getY() + sideLength)
        );

        if (sideLength <= 0) {
            throw new IllegalArgumentException(
                "Side length must be positive"
            );
        }
    }
}

public class Practical1 {

    public static void main(String[] args) {

        Point p1 = new Point(0, 0);
        Point p2 = new Point(4, 0);
        Point p3 = new Point(4, 3);
        Point p4 = new Point(0, 3);

        Trapezoid trapezoid =
            new Trapezoid(p1, p2, p3, p4);

        Rectangle rectangle =
            new Rectangle(p1, p2, p3, p4);

        Square square = new Square(p1, 3);

        System.out.println(
            "Area of trapezoid = " + trapezoid.calculateArea()
        );

        System.out.println(
            "Area of rectangle = " + rectangle.calculateArea()
        );

        System.out.println(
            "Area of square = " + square.calculateArea()
        );
    }
}
